## 4. 完整 i2v_polling.py（后台无人值守）

```python
#!/usr/bin/env python3
"""稳健的 dreamina i2v 串行 polling"""
import subprocess, re, time, os, urllib.request
from datetime import datetime

PROJ = os.path.expanduser('~/Downloads/AI短剧_<剧名>')
SHOTS = f'{PROJ}/04_单镜头生图/第一集'
VIDS = f'{PROJ}/05_单镜头生视频/第一集'
LOG = '/tmp/i2v_polling.log'

def log(msg):
    line = f'[{datetime.now().strftime("%H:%M:%S")}] {msg}'
    print(line, flush=True)
    open(LOG, 'a').write(line + '\n')

TASKS = [
    # (id, image_filename, prompt, output_filename)
    ('3', '镜头3.png', '镜头3 的 image2video prompt...', '镜头3.mp4'),
    # ... 其他 13 个
]

def extract_uuid(text):
    m = re.search(r'"submit_id"\s*:\s*"([a-f0-9-]{36})"', text)
    return m.group(1) if m else None

def query_status(sid):
    out = subprocess.run(['dreamina', 'query_result', '--submit_id', sid],
                         capture_output=True, text=True, timeout=60).stdout
    url_m = re.search(r'"video_url"\s*:\s*"([^"]+)"', out)
    if url_m:
        return ('done', url_m.group(1))
    status_m = re.search(r'"gen_status"\s*:\s*"([^"]+)"', out)
    queue_m = re.search(r'"queue_idx"\s*:\s*(\d+)', out)
    return ('pending', f'status={status_m.group(1) if status_m else "?"} queue={queue_m.group(1) if queue_m else "-"}')

def submit(image_path, prompt):
    for retry in range(3):
        try:
            r = subprocess.run([
                'dreamina', 'image2video',
                '--image', image_path,
                '--prompt', prompt,
                '--duration', '5',
                '--video_resolution', '720p',
                '--model_version', 'seedance2.0fast',
                '--poll', '10'
            ], capture_output=True, text=True, timeout=120)
            sid = extract_uuid(r.stdout + r.stderr)
            if sid:
                if '"gen_status"' in r.stdout and '"fail"' in r.stdout:
                    log(f'  submit returned fail, retry {retry+1}')
                    time.sleep(60)
                    continue
                return sid
        except Exception as e:
            log(f'  submit error: {e}')
        time.sleep(30)
    return None

def wait_for_completion(sid, label, max_polls=80):
    for i in range(max_polls):
        kind, info = query_status(sid)
        if kind == 'done':
            return info
        log(f'  {label}: {info} (poll {i+1}/{max_polls})')
        time.sleep(30)
    return None

def main():
    for sid_label, img_name, prompt, out_name in TASKS:
        time.sleep(30)  # rate limit buffer
        log(f'=== Submitting 镜头{sid_label} ===')
        img_path = f'{SHOTS}/{img_name}'
        if not os.path.exists(img_path):
            log(f'  WARNING: {img_path} not found, skip')
            continue
        sid = submit(img_path, prompt)
        if not sid:
            log(f'  镜头{sid_label} submit failed, skip')
            continue
        log(f'  镜头{sid_label} sid={sid}')
        url = wait_for_completion(sid, f'镜头{sid_label}')
        if url:
            out = f'{VIDS}/{out_name}'
            urllib.request.urlretrieve(url, out)
            log(f'  saved [{os.path.getsize(out)} bytes]: {out_name}')

    log('=== ALL DONE ===')

if __name__ == '__main__':
    main()
```

**启动**：`nohup python3 /tmp/i2v_polling.py > /tmp/i2v_polling.stdout 2>&1 &`

## 5. watchdog_v4.sh（自动调起 master）

```bash
#!/bin/bash
LOG=/tmp/watchdog_v4.log
echo "[$(date '+%H:%M:%S')] watchdog start" > "$LOG"

MAX_WAIT=7200  # 2 小时硬封顶
elapsed=0
while [ $elapsed -lt $MAX_WAIT ]; do
  if grep -q "ALL DONE" /tmp/i2v_polling.log 2>/dev/null; then
    echo "[$(date '+%H:%M:%S')] polling done, starting master" >> "$LOG"
    break
  fi
  if ! pgrep -f i2v_polling.py > /dev/null 2>&1; then
    echo "[$(date '+%H:%M:%S')] polling exited" >> "$LOG"
    break
  fi
  sleep 60
  elapsed=$((elapsed + 60))
done

bash /tmp/master_v4.sh 2>&1 | tee -a "$LOG"
```

```bash

# Master v4 发布级 pipeline
set -e

PROJ="$HOME/Downloads/AI短剧_<剧名>"
SHOTS="$PROJ/04_单镜头生图/第一集"
VIDS="$PROJ/05_单镜头生视频/第一集"

rm -rf "$WORK"; mkdir -p "$WORK" "$OUT_DIR"

LOG="/tmp/master_v4.log"
log() { echo "[$(date '+%H:%M:%S')] $*" | tee -a "$LOG"; }

# ============ STEP 1: 每镜头出标准化 clip ============
SHOT_LIST=(
  "1|5|镜头1_废墟.png|镜头1.mp4"
  "2|5|镜头2_庄园.png|镜头2.mp4"
  "3|3|镜头3.png|镜头3.mp4"
  # ... 14 镜头
)

CONCAT="$WORK/concat.txt"
> "$CONCAT"

for entry in "${SHOT_LIST[@]}"; do
  IFS='|' read -r id dur img vid <<< "$entry"
  out_clip="$WORK/clip_$(printf '%02d' $id).mp4"
  COLOR="eq=contrast=1.05:saturation=1.12:gamma=0.98,vignette=PI/5"
  FADE="fade=t=in:st=0:d=0.3,fade=t=out:st=$(echo "$dur-0.3" | bc):d=0.3"
  if [ -n "$vid" ] && [ -f "$VIDS/$vid" ]; then
    # 用视频
    ffmpeg -y -i "$VIDS/$vid" -t "$dur" \
      -vf "scale=1280:720:force_original_aspect_ratio=decrease,pad=1280:720:(ow-iw)/2:(oh-ih)/2:black,setsar=1,fps=30,${COLOR},${FADE}" \
      -c:v libx264 -preset fast -crf 20 -pix_fmt yuv420p -an "$out_clip" 2>/dev/null
  else
    # 静帧 fallback
    frames=$((dur * 30))
    ffmpeg -y -loop 1 -i "$SHOTS/$img" -t "$dur" \
      -vf "scale=1920:1080:force_original_aspect_ratio=increase,crop=1920:1080,zoompan=z='min(zoom+0.001,1.10)':d=${frames}:x='iw/2-(iw/zoom/2)':y='ih/2-(ih/zoom/2)':s=1280x720,setsar=1,fps=30,${COLOR},${FADE}" \
      -c:v libx264 -preset fast -crf 20 -pix_fmt yuv420p -an "$out_clip" 2>/dev/null
  fi
  echo "file '$out_clip'" >> "$CONCAT"
done

# ============ STEP 2: 片头+14镜头+片尾 拼接 ============
INTRO="$AUDIO_DIR/intro.mp4"
OUTRO="$AUDIO_DIR/outro.mp4"
CONCAT_FULL="$WORK/concat_full.txt"
echo "file '$INTRO'" > "$CONCAT_FULL"
cat "$CONCAT" >> "$CONCAT_FULL"
echo "file '$OUTRO'" >> "$CONCAT_FULL"

RAW_VID="$WORK/v4_raw.mp4"
ffmpeg -y -f concat -safe 0 -i "$CONCAT_FULL" -c copy "$RAW_VID" 2>/dev/null

# ============ STEP 3: drawtext 烧中文字幕 (15 段链式) ============
FONT='/System/Library/Fonts/STHeiti Medium.ttc'
COMMON="fontfile=${FONT// /\\ }:shadowcolor=black:shadowx=2:shadowy=2:borderw=2:bordercolor=black:x=(w-text_w)/2:y=h-100"

WITH_SUB="$WORK/v4_with_sub.mp4"
ffmpeg -y -i "$RAW_VID" -filter_complex "
[0:v]drawtext=${COMMON}:fontsize=34:fontcolor=#FFFFFF:textfile=/tmp/subs/01.txt:enable='between(t,4,7)'[v1];
[v1]drawtext=${COMMON}:fontsize=36:fontcolor=#FF1515:textfile=/tmp/subs/02.txt:enable='between(t,7,9)'[v2];
[v2]drawtext=${COMMON}:fontsize=36:fontcolor=#FFDD00:textfile=/tmp/subs/03.txt:enable='between(t,9.5,14)'[v3];
... (15 段链式)
[v14]drawtext=${COMMON}:fontsize=34:fontcolor=#00FF00:textfile=/tmp/subs/15.txt:enable='between(t,56.5,61)'[vout]
" -map "[vout]" -c:v libx264 -preset fast -crf 20 -pix_fmt yuv420p -an "$WITH_SUB" 2>&1 | tail -3

# ============ STEP 4: 混音 — BGM + 15 TTS + 4 SFX = 20 amix ============
ffmpeg -y -i "$WITH_SUB" \
  -i "$AUDIO_DIR/bgm/bgm_full.wav" \
  -i "$AUDIO_DIR/01a.mp3" -i "$AUDIO_DIR/01b.mp3" \
  -i "$AUDIO_DIR/02.mp3"  ... -i "$AUDIO_DIR/14.mp3" \
  -i "$AUDIO_DIR/sfx/thunder.wav" \
  -i "$AUDIO_DIR/sfx/woosh.wav" \
  -i "$AUDIO_DIR/sfx/kick.wav" \
  -i "$AUDIO_DIR/sfx/thud.wav" \
  -filter_complex "
[1:a]aresample=44100,adelay=4000|4000,volume=0.18[bgm];
[2:a]aresample=44100,adelay=4500|4500,volume=14[t1];
[3:a]aresample=44100,adelay=7000|7000,volume=14[t2];
...
[17:a]aresample=44100,adelay=42000|42000,volume=8[sfx1];
[18:a]aresample=44100,adelay=11000|11000,volume=8[sfx2];
[19:a]aresample=44100,adelay=31000|31000,volume=8[sfx3];
[20:a]aresample=44100,adelay=21000|21000,volume=6[sfx4];
[bgm][t1][t2]...[sfx4]amix=inputs=20:dropout_transition=0[aout]" \
  -map 0:v -map "[aout]" \
  -c:v copy -c:a aac -b:a 192k -shortest \
  "$OUT_DIR/第一集_成片_v4_发布级.mp4" 2>&1 | tail -5

# ============ STEP 5: 9:16 竖版 ============
V_OUT="$OUT_DIR/第一集_成片_v4_竖版.mp4"
ffmpeg -y -i "$OUT_DIR/第一集_成片_v4_发布级.mp4" \
  -vf "scale=720:-1,pad=720:1280:0:(1280-ih)/2:black" \
  -c:v libx264 -preset fast -crf 20 -pix_fmt yuv420p \
  -c:a copy "$V_OUT" 2>&1 | tail -3

log "=== Master v4 ALL DONE ==="
```
